    async function initWebcamProctoring() {
      const video = document.getElementById('webcamFeed');
      if (!video) return;

      try {
        const stream = await navigator.mediaDevices.getUserMedia({ 
          video: { width: { ideal: 320 }, height: { ideal: 240 } } 
        });
        
        video.srcObject = stream;
        await video.play().catch(function(e) { console.warn("Video playback interrupted:", e); });

        if (!cocoModel) {
          cocoModel = await cocoSsd.load();
        }
        
        if (proctorInterval) clearInterval(proctorInterval);
        proctorInterval = setInterval(detectPhone, 2000);
      } catch (err) {
        console.warn("Webcam access error or permission denied:", err);
        showBootstrapToast("Proctor Alert: Camera access is disabled or unavailable.", "warning");
      }
    }

    async function detectPhone() {
      const video = document.getElementById('webcamFeed');
      if (!cocoModel || !video || !isTestActive) return;

      if (video.readyState < 2 || video.paused || video.ended || video.videoWidth === 0) {
        return;
      }

      try {
        const predictions = await cocoModel.detect(video);
        
        predictions.forEach(prediction => {
          if (prediction.class === 'cell phone' && prediction.score > 0.6) {
            showBootstrapToast("SECURITY ALERT: Mobile phone detected by proctoring camera!", "danger");
            pasteEventCount += 2; 
          }
        });
      } catch (err) {
        console.error("COCO-SSD Detection error:", err);
      }
    }

    function processUpload() {
      var startDateVal = document.getElementById('startDatePicker').value;
      var startTimeVal = document.getElementById('startTimePicker').value || '08:00';

      var dateVal = document.getElementById('expiryDatePicker').value;
      var timeVal = document.getElementById('expiryTimePicker').value || '23:59';
      var timeoutMinsVal = document.getElementById('timeoutMinsPicker').value || 60;
      
      if (!dateVal) {
        showBootstrapToast("Please select an Expiration Date!", "warning");
        return;
      }

      var displayStartVal = startDateVal ? (startDateVal + " " + startTimeVal) : "";
      var displayUntilVal = dateVal + " " + timeVal;
      var uploaderName = (currentUserData && currentUserData.name) ? currentUserData.name : 'Faculty User';
      var fileInput = document.getElementById('excelFileInput');

      var parsedQuestions = [];

      if (fileInput.files.length > 0) {
        var file = fileInput.files[0];
        var reader = new FileReader();

        reader.onload = function(e) {
          try {
            var data = new Uint8Array(e.target.result);
            var workbook = XLSX.read(data, { type: 'array' });
            var firstSheet = workbook.Sheets[workbook.SheetNames[0]];
            var rawRows = XLSX.utils.sheet_to_json(firstSheet, { header: 1 });

            var startRow = 0;
            if (rawRows.length > 0 && String(rawRows[0][0]).toLowerCase().indexOf('question') !== -1) {
              startRow = 1;
            }

            for (var i = startRow; i < rawRows.length; i++) {
              var row = rawRows[i];
              if (row && row[0]) {
                var rawType = String(row[1] || "").trim().toUpperCase();
                var isCompiler = (rawType === "COMPILER") || (!row[2] && !row[3]);

                parsedQuestions.push({
                  question: row[0],
                  type: isCompiler ? "COMPILER" : "MCQ",
                  optionA: isCompiler ? '' : (row[2] || ''),
                  optionB: isCompiler ? '' : (row[3] || ''),
                  optionC: isCompiler ? '' : (row[4] || ''),
                  optionD: isCompiler ? '' : (row[5] || ''),
                  sampleInput: isCompiler ? (row[2] || '') : '',
                  sampleOutput: isCompiler ? (row[3] || '') : '',
                  answer: row[6] || '',
                  marks: row[7] || 1,
                  hiddenTestCases: isCompiler ? (row[6] || '') : '',
                  uploadedBy: uploaderName
                });
              }
            }

            executeUploadServerCall(parsedQuestions, displayStartVal, displayUntilVal, timeoutMinsVal);
          } catch (err) {
            showBootstrapToast("Error parsing Excel: " + err.message, "danger");
          }
        };

        reader.readAsArrayBuffer(file);
      } else { 
        var q = document.getElementById('mQ').value;
        if (!q) { showBootstrapToast("Please upload an Excel file or type a question.", "warning"); return; }
        
        var mType = document.getElementById('mType').value;
        parsedQuestions.push({
          question: q,
          type: mType,
          optionA: mType === "MCQ" ? document.getElementById('mA').value : '',
          optionB: mType === "MCQ" ? document.getElementById('mB').value : '',
          optionC: mType === "MCQ" ? document.getElementById('mC').value : '',
          optionD: mType === "MCQ" ? document.getElementById('mD').value : '',
          sampleInput: mType === "COMPILER" ? document.getElementById('mA').value : '',
          sampleOutput: mType === "COMPILER" ? document.getElementById('mB').value : '',
          answer: document.getElementById('mAns').value,
          marks: document.getElementById('mMarks').value || 1,
          hiddenTestCases: mType === "COMPILER" ? document.getElementById('mAns').value : '',
          uploadedBy: uploaderName
        });
        executeUploadServerCall(parsedQuestions, displayStartVal, displayUntilVal, timeoutMinsVal);
      }
    }

    function executeUploadServerCall(questionsArray, displayStartVal, displayUntilVal, timeoutMinsVal) {

            showBootstrapToast("Published " + res.count + " questions successfully into " + res.targetSheet, "success");
          } else {
            showBootstrapToast("Upload Error: " + (res ? res.message : "Unknown error"), "danger");
          }
        })
        .uploadQuestionsBatch(selectedUploadType, questionsArray, displayStartVal, displayUntilVal, timeoutMinsVal);
    }

    function startExamTimer(durationMins) {
      totalExamSeconds = (durationMins || 60) * 60;
      clearInterval(examTimerInterval);
      
      examTimerInterval = setInterval(function() {
        if (totalExamSeconds <= 0) {
          clearInterval(examTimerInterval);
          showBootstrapToast("Assessment Timeout! Time has expired. Submitting your test automatically...", "warning");
          confirmAndExecuteSubmit(true);
          return;
        }

        totalExamSeconds--;
        var mins = Math.floor(totalExamSeconds / 60);
        var secs = totalExamSeconds % 60;
        document.getElementById('examTimerDisplay').innerText = 
          (mins < 10 ? "0" : "") + mins + ":" + (secs < 10 ? "0" : "") + secs;
      }, 1000);
    }

    function renderQuestion() {
      var q = questionsList[currentQIndex];
      var total = questionsList.length;
      document.getElementById('questionProgressText').innerText = "Question " + (currentQIndex + 1) + " of " + total;
      document.getElementById('questionText').innerText = (currentQIndex + 1) + ". " + q.question;

      var aptitudeContainer = document.getElementById('aptitudeModeContainer');
      var codingContainer = document.getElementById('codingModeContainer');
      var optionsContainer = document.getElementById('optionsContainer');

      if (q.type === 'MCQ') {
        aptitudeContainer.classList.remove('d-none');
        aptitudeContainer.classList.add('d-flex');
        codingContainer.style.display = 'none';

        optionsContainer.innerHTML = '';
        ['A', 'B', 'C', 'D'].forEach(function(lbl, idx) {
          var opt = q.options[idx];
          if (!opt) return;
          var checked = userAnswers[q.id] === opt ? 'checked' : '';
          optionsContainer.innerHTML += 
            '<label class="option-card" onclick="selectOption(\'' + q.id + '\', \'' + opt + '\')">' +
              '<input type="radio" name="opt_' + q.id + '" ' + checked + '>' +
              '<div><strong>' + lbl + '.</strong> ' + opt + '</div>' +
            '</label>';
        });
      } else {
        aptitudeContainer.classList.remove('d-flex');
        aptitudeContainer.classList.add('d-none');
        optionsContainer.innerHTML = '';

        codingContainer.style.display = 'block';

        var sampleInContainer = document.getElementById('sampleInputContainer');
        var sampleInBox = document.getElementById('sampleInputBox');
        if (q.sampleInput && q.sampleInput.trim() !== '') {
          sampleInContainer.style.display = 'block';
          sampleInBox.innerText = q.sampleInput;
        } else {
          sampleInContainer.style.display = 'none';
        }

        var sampleOutContainer = document.getElementById('sampleOutputContainer');
        var sampleOutBox = document.getElementById('sampleOutputBox');
        if (q.sampleOutput && q.sampleOutput.trim() !== '') {
          sampleOutContainer.style.display = 'block';
          sampleOutBox.innerText = q.sampleOutput;
        } else {
          sampleOutContainer.style.display = 'none';
        }

        initMonacoEditor();
        var savedCode = userAnswers[q.id] || codeTemplates[document.getElementById('languageSelect').value];
        if (monacoEditorInstance) {
          monacoEditorInstance.setValue(savedCode);
        }
      }

      document.getElementById('prevBtn').disabled = currentQIndex === 0;
      document.getElementById('nextBtn').style.display = (currentQIndex === total - 1) ? 'none' : 'inline-block';
    }

    function selectOption(qId, val) {
      userAnswers[qId] = val;
      renderQuestionPalette();
    }

    function renderQuestionPalette() {
      var grid = document.getElementById('questionPaletteGrid');
      grid.innerHTML = '';

      questionsList.forEach(function(q, idx) {
        var isAnswered = userAnswers[q.id] !== undefined && String(userAnswers[q.id]).trim() !== '';
        var isCurrent = idx === currentQIndex;

        var btnClass = 'palette-unanswered';
        if (isAnswered) btnClass = 'palette-answered';
        if (isCurrent) btnClass = 'palette-current';

        grid.innerHTML += 
          '<button class="btn palette-btn ' + btnClass + '" onclick="jumpToQuestion(' + idx + ')">' +
            (idx + 1) +
          '</button>';
      });
    }

    function jumpToQuestion(index) {
      saveCurrentCodeState();
      currentQIndex = index;
      renderQuestion();
      renderQuestionPalette();
    }

    function nextQuestion() {
      saveCurrentCodeState();
      currentQIndex++;
      renderQuestion();
      renderQuestionPalette();
    }

    function prevQuestion() {
      saveCurrentCodeState();
      currentQIndex--;
      renderQuestion();
      renderQuestionPalette();
    }

    function saveCurrentCodeState() {
      var q = questionsList[currentQIndex];
      if (q && q.type === 'COMPILER' && monacoEditorInstance) {
        userAnswers[q.id] = monacoEditorInstance.getValue();
      }
    }

    function attemptFinalSubmit(isTimerExpired) {
      saveCurrentCodeState();

      if (!isTimerExpired) {
        var unansweredList = [];
        questionsList.forEach(function(q, index) {
          var ans = userAnswers[q.id];
          if (!ans || String(ans).trim() === '') {
            unansweredList.push(index + 1);
          }
        });

        if (unansweredList.length > 0) {
          showBootstrapToast("Submission Blocked: You must answer all questions before submitting! Unanswered question(s): Q" + unansweredList.join(', Q'), "danger");
          return;
        }

        var confirmTextEl = document.getElementById('submitConfirmText');
        confirmTextEl.innerText = "All questions have been answered. Are you sure you want to submit your test?";

        if (typeof bootstrap !== 'undefined') {
          new bootstrap.Modal(document.getElementById('submitConfirmModal')).show();
        }
      } else {
        confirmAndExecuteSubmit(true);
      }
    }

    function confirmAndExecuteSubmit(isTimerExpired) {
      clearInterval(examTimerInterval);
      if (proctorInterval) clearInterval(proctorInterval);

      const video = document.getElementById('webcamFeed');
      if (video && video.srcObject) {
        video.srcObject.getTracks().forEach(track => track.stop());
      }

      if (typeof bootstrap !== 'undefined') {
        var modalEl = document.getElementById('submitConfirmModal');
        var modalInstance = bootstrap.Modal.getInstance(modalEl);
        if (modalInstance) modalInstance.hide();
      }

      document.getElementById('testContainer').style.display = 'none';
      showLoading("Auditing & Submitting...", "Running Proctoring Security Agent...");
      isTestActive = false;

      var durationSec = testStartTime ? Math.floor((new Date() - testStartTime) / 1000) : 0;

      google.script.run
        .withSuccessHandler(function(auditRes) {
          google.script.run
            .withSuccessHandler(function(res) {
              hideLoading();
              if (res && res.success) {
                showBootstrapToast("Successfully Completed", "success", true);
              } else {
                showBootstrapToast("Error: " + (res ? res.message : "Submission failed"), "danger");
              }
            })
            .submitTestAnswers(userAnswers, isTimerExpired || false);
        })
        .runProctoringAgent(pasteEventCount, tabSwitchCount, durationSec);
    }

    function initMonacoEditor() {
      if (!monacoEditorInstance && typeof require !== 'undefined') {
        require.config({ paths: { vs: 'https://cdnjs.cloudflare.com/ajax/libs/monaco-editor/0.39.0/min/vs' } });
        require(['vs/editor/editor.main'], function () {
          monacoEditorInstance = monaco.editor.create(document.getElementById('monaco-container'), {
            value: codeTemplates.python,
            language: 'python',
            theme: 'vs-dark',
            automaticLayout: true,
            fontSize: 14,
            minimap: { enabled: false }
          });
          
          monacoEditorInstance.onDidChangeModelContent(function() {
            var currentQ = questionsList[currentQIndex];
            if (currentQ) {
              userAnswers[currentQ.id] = monacoEditorInstance.getValue();
              renderQuestionPalette();
            }
          });
        });
      }
    }

    function changeLanguage() {
      var lang = document.getElementById('languageSelect').value;
      if (monacoEditorInstance) {
        monaco.editor.setModelLanguage(monacoEditorInstance.getModel(), lang === 'cpp' ? 'cpp' : lang);
        monacoEditorInstance.setValue(codeTemplates[lang] || '');
      }
    }

    function runCodeAgainstPiston() {
      var consoleEl = document.getElementById('consoleOutput');
      var selectedLangKey = document.getElementById('languageSelect').value;
      var code = monacoEditorInstance ? monacoEditorInstance.getValue() : '';
      var languageId = judge0LangMap[selectedLangKey];

      if (!code.trim()) { consoleEl.innerText = "Error: Editor is empty!"; return; }

      consoleEl.innerText = "Compiling and executing code...";
      fetch('https://ce.judge0.com/submissions?wait=true', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ source_code: code, language_id: languageId, stdin: "" })
      })
      .then(function(res) { return res.json(); })
      .then(function(data) {
        consoleEl.innerText = data.stdout || data.stderr || data.compile_output || "Executed with no output.";
      })
      .catch(function(err) { consoleEl.innerText = "Execution API Error: " + err.message; });
    }

    function toggleChatWindow() {
      var win = document.getElementById('chatbotWindow');
      win.style.display = (win.style.display === 'flex') ? 'none' : 'flex';
    }

    function sendChatMessage() {
      var inputEl = document.getElementById('chatInput');
      var prompt = inputEl.value.trim();
      if (!prompt) return;

      var chatBody = document.getElementById('chatBody');

      chatBody.innerHTML += '<div class="chat-msg msg-user">' + escapeHtml(prompt) + '</div>';
      inputEl.value = '';
      chatBody.scrollTop = chatBody.scrollHeight;

      var loadingId = 'loading_' + Date.now();
      chatBody.innerHTML += '<div id="' + loadingId + '" class="chat-msg msg-bot text-info"><i class="fa-solid fa-circle-notch fa-spin me-2"></i>Thinking...</div>';
      chatBody.scrollTop = chatBody.scrollHeight;

      var currentQContext = "";
      if (typeof questionsList !== 'undefined' && questionsList[currentQIndex]) {
        currentQContext = questionsList[currentQIndex].question;
      }

      google.script.run
        .withSuccessHandler(function(res) {
          var loader = document.getElementById(loadingId);
          if (loader) loader.remove();

          if (res && res.success) {
            chatBody.innerHTML += '<div class="chat-msg msg-bot">' + escapeHtml(res.reply) + '</div>';
          } else {
            chatBody.innerHTML += '<div class="chat-msg msg-bot text-danger">Error: ' + escapeHtml(res ? res.message : "Failed to generate reply") + '</div>';
          }
          chatBody.scrollTop = chatBody.scrollHeight;
        })
        .withFailureHandler(function(err) {
          var loader = document.getElementById(loadingId);
          if (loader) loader.remove();
          chatBody.innerHTML += '<div class="chat-msg msg-bot text-danger">Server Error: ' + escapeHtml(err.toString()) + '</div>';
          chatBody.scrollTop = chatBody.scrollHeight;
        })
        .callGroqChatbot(prompt, isTestActive || false, currentQContext);
    }

    function escapeHtml(text) {
      return text.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;").replace(/'/g, "&#039;");
    }
